// Optional constraints (safe to re-run)
CREATE CONSTRAINT cwe_id_unique IF NOT EXISTS
FOR (c:CWE) REQUIRE c.cwe_id IS UNIQUE;

CREATE CONSTRAINT cwe_mitigation_id_unique IF NOT EXISTS
FOR (m:CWEMitigation) REQUIRE m.mitigation_id IS UNIQUE;

// Create relationships from file placed in Neo4j's import dir
LOAD CSV WITH HEADERS FROM 'file:///HAS_CWE_MITIGATION.csv' AS row
WITH row
CALL {
  WITH row
  WITH trim(row.cwe_id) AS cweId, trim(row.id) AS mitId
  WHERE cweId <> '' AND mitId <> ''

  // Match existing nodes
  MATCH (w:CWE) WHERE toLower(w.cwe_id) = toLower(cweId)
  MATCH (m:CWEMitigation {mitigation_id: mitId})

  // Relationship name = file name
  MERGE (w)-[:HAS_CWE_MITIGATION]->(m)
} IN TRANSACTIONS OF 1000 ROWS;
