///////////////////////////////////////////////////////////////////////////
// (Recommended) indexes for fast matching
///////////////////////////////////////////////////////////////////////////
CREATE CONSTRAINT IF NOT EXISTS
FOR (p:Product) REQUIRE p.product_id IS UNIQUE;

CREATE CONSTRAINT IF NOT EXISTS
FOR (h:Host) REQUIRE h.id IS UNIQUE;

CREATE CONSTRAINT IF NOT EXISTS
FOR (a:Account) REQUIRE a.id IS UNIQUE;

CREATE INDEX comp_machine_id IF NOT EXISTS
FOR (c:Component) ON (c.machine_id);

///////////////////////////////////////////////////////////////////////////
// 1) AUTHENTICATES.csv  ->  (:Account)-[:AUTHENTICATES]->(:Host|:Product)
///////////////////////////////////////////////////////////////////////////
LOAD CSV WITH HEADERS FROM 'file:///AUTHENTICATES.csv' AS row
WITH trim(row.account_id) AS accId,
     toLower(trim(row.target_kind)) AS kind,
     trim(row.target_id) AS tgt,
     row.identity AS ident,
     row.use AS useVal
WHERE accId <> '' AND tgt <> ''
CALL {
  WITH accId, kind, tgt, ident, useVal
  MATCH (a:Account {id: accId})

  // Host target (match by id or ip)
  CALL {
    WITH kind, tgt
    WITH kind, tgt
    WHERE kind CONTAINS 'host'
    MATCH (h:Host)
      WHERE h.id = tgt OR h.ip = tgt
    RETURN h
  } CALL {
    WITH kind, tgt
    WITH kind, tgt
    WHERE kind IN ['device','product','asset'] OR kind CONTAINS 'device' OR kind CONTAINS 'product' OR kind CONTAINS 'asset'
    MATCH (p:Product {product_id: tgt})
    RETURN p
  }
  WITH a, h, p, ident, useVal
  // Merge to whichever matched
  FOREACH (_ IN CASE WHEN h IS NOT NULL THEN [1] ELSE [] END |
    MERGE (a)-[r:AUTHENTICATES]->(h)
    SET r.identity = CASE WHEN ident IS NULL OR trim(ident) = '' THEN r.identity ELSE ident END,
        r.use      = CASE WHEN useVal IS NULL OR trim(useVal) = '' THEN r.use      ELSE useVal END
  )
  FOREACH (_ IN CASE WHEN p IS NOT NULL THEN [1] ELSE [] END |
    MERGE (a)-[r:AUTHENTICATES]->(p)
    SET r.identity = CASE WHEN ident IS NULL OR trim(ident) = '' THEN r.identity ELSE ident END,
        r.use      = CASE WHEN useVal IS NULL OR trim(useVal) = '' THEN r.use      ELSE useVal END
  )
} IN TRANSACTIONS OF 20000 ROWS;

///////////////////////////////////////////////////////////////////////////
// 2) HOST_ASSET.csv  ->  (:Product)-[:HOST_ASSET]->(:Host)
///////////////////////////////////////////////////////////////////////////
LOAD CSV WITH HEADERS FROM 'file:///HOST_ASSET.csv' AS row
WITH trim(row.id) AS productId, trim(row.ip) AS ip
WHERE productId <> '' AND ip <> ''
CALL {
  WITH productId, ip
  MATCH (p:Product {product_id: productId})
  MATCH (h:Host) WHERE h.id = ip OR h.ip = ip
  MERGE (p)-[:HOST_ASSET]->(h)
} IN TRANSACTIONS OF 20000 ROWS;

///////////////////////////////////////////////////////////////////////////
// 3) COMMUNICATES_WITH.csv  ->  (:Product)-[:COMMUNICATES_WITH {flow_id}]->(:Product)
///////////////////////////////////////////////////////////////////////////
LOAD CSV WITH HEADERS FROM 'file:///COMMUNICATES_WITH.csv' AS row
WITH trim(row.flow_id) AS flow,
     trim(row.src_asset_id) AS srcId,
     trim(row.dst_asset_id) AS dstId,
     CASE WHEN row.protocol IS NULL THEN NULL ELSE toUpper(trim(row.protocol)) END AS proto,
     CASE WHEN row.port     IS NULL OR trim(row.port) = '' THEN NULL ELSE toInteger(row.port) END AS portNum,
     row.first_ts AS firstTs,
     row.last_ts  AS lastTs,
     CASE WHEN row.bytes    IS NULL OR trim(row.bytes) = '' THEN NULL ELSE toInteger(row.bytes) END AS bytesNum,
     CASE WHEN row.riskWeight IS NULL OR trim(row.riskWeight) = '' THEN NULL ELSE toFloat(row.riskWeight) END AS riskW
WHERE flow <> '' AND srcId <> '' AND dstId <> ''
CALL {
  WITH flow, srcId, dstId, proto, portNum, firstTs, lastTs, bytesNum, riskW
  MATCH (s:Product {product_id: srcId})
  MATCH (d:Product {product_id: dstId})
  MERGE (s)-[r:COMMUNICATES_WITH {flow_id: flow}]->(d)
  SET r.protocol  = proto,
      r.port      = portNum,
      r.first_ts  = firstTs,
      r.last_ts   = lastTs,
      r.bytes     = bytesNum,
      r.riskWeight= riskW
} IN TRANSACTIONS OF 20000 ROWS;

///////////////////////////////////////////////////////////////////////////
// 4) HAS_COMPONENT.csv  ->  (:Product)-[:HAS_COMPONENT]->(:Component)
///////////////////////////////////////////////////////////////////////////
LOAD CSV WITH HEADERS FROM 'file:///HAS_COMPONENT.csv' AS row
WITH trim(row.asset_id) AS productId, trim(row.machine_id) AS compKey
WHERE productId <> '' AND compKey <> ''
CALL {
  WITH productId, compKey
  MATCH (p:Product {product_id: productId})
  MATCH (c:Component {machine_id: compKey})
  MERGE (p)-[:HAS_COMPONENT]->(c)
} IN TRANSACTIONS OF 20000 ROWS;

LOAD CSV WITH HEADERS FROM 'file:///CONTROLS.csv' AS row
WITH trim(row.controller_id) AS ctrlId,
     toLower(trim(row.controller_type)) AS ctrlType,
     trim(row.component_id) AS compId,
     row.actuation AS actuation
WHERE ctrlId <> '' 
  AND compId <> '' 
  AND (ctrlType CONTAINS 'product' OR ctrlType CONTAINS 'asset')
  AND NOT (ctrlType CONTAINS 'device')
CALL {
  WITH ctrlId, compId, actuation
  MATCH (p:Product  {product_id: ctrlId})
  MATCH (c:Component {machine_id: compId})
  MERGE (p)-[r:CONTROLS]->(c)
  SET r.actuation = CASE WHEN actuation IS NULL OR trim(actuation) = '' THEN r.actuation ELSE actuation END
} IN TRANSACTIONS OF 20000 ROWS;
