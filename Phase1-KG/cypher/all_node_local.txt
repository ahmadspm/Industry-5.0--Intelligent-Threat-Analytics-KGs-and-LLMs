///////////////////////////////////////////////////////////////////////////
// CONSTRAINTS (safe to re-run)
///////////////////////////////////////////////////////////////////////////

// component: no single id column; key by (machine_id, name)
CREATE CONSTRAINT IF NOT EXISTS
FOR (c:Component) REQUIRE (c.machine_id, c.name) IS NODE KEY;

CREATE CONSTRAINT IF NOT EXISTS
FOR (m:Machine) REQUIRE m.id IS UNIQUE;

CREATE CONSTRAINT IF NOT EXISTS
FOR (h:Host) REQUIRE h.id IS UNIQUE;

CREATE CONSTRAINT IF NOT EXISTS
FOR (a:Alert) REQUIRE a.id IS UNIQUE;

CREATE CONSTRAINT IF NOT EXISTS
FOR (e:Event) REQUIRE e.id IS UNIQUE;

CREATE CONSTRAINT IF NOT EXISTS
FOR (cr:Credential) REQUIRE cr.id IS UNIQUE;

CREATE CONSTRAINT IF NOT EXISTS
FOR (ac:Account) REQUIRE ac.id IS UNIQUE;

CREATE CONSTRAINT IF NOT EXISTS
FOR (s:Sensor) REQUIRE s.id IS UNIQUE;

CREATE CONSTRAINT IF NOT EXISTS
FOR (pv:ProcessVariable) REQUIRE pv.id IS UNIQUE;

CREATE CONSTRAINT IF NOT EXISTS
FOR (act:Actuator) REQUIRE act.id IS UNIQUE;

///////////////////////////////////////////////////////////////////////////
// LOADERS (batched; no USING PERIODIC COMMIT)
///////////////////////////////////////////////////////////////////////////

// -------------------- Component --------------------
LOAD CSV WITH HEADERS FROM 'file:///component_node.csv' AS row
WITH row
CALL {
  WITH row
  WITH
    trim(row.machine_id) AS machine_id,
    trim(row.name)       AS name,
    row.kind             AS kindRaw,
    row.safety_level     AS safetyRaw
  WHERE machine_id <> '' AND name <> ''

  MERGE (c:Component {machine_id: machine_id, name: name})
  SET c.kind         = CASE WHEN kindRaw   IS NULL OR trim(kindRaw)   = '' THEN NULL ELSE kindRaw   END,
      c.safety_level = CASE WHEN safetyRaw IS NULL OR trim(safetyRaw) = '' THEN NULL ELSE safetyRaw END
} IN TRANSACTIONS OF 1000 ROWS;

// -------------------- Machine --------------------
LOAD CSV WITH HEADERS FROM 'file:///machine_node.csv' AS row

    trim(row.id)        AS id,

    row.family          AS family,

      m.family     = CASE WHEN family     IS NULL OR trim(family)     = '' THEN NULL ELSE family     END,
